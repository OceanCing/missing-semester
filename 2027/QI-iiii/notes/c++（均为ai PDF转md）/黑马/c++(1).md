4. 运算符

算术运算符

`+` 加
`-` 减
`*` 乘
`/` 除
`%` 取余

```cpp
int a1 = 10;
int b1 = a1 + 2;      // b1 = 12

int a2 = 10;
int b2 = ++a2 * 10;   // b2 = 110，a2 = 11

int a3 = 10;
int b3 = a3++ * 10;   // b3 = 100，a3 = 11
```

前置递增：先让变量加 `1` 再参与运算
后置递增：先参与运算 再让变量加 `1`

赋值运算符

```cpp
a = 10;       // a = a + 2

a += 2;       // a = 12
```

比较运算符

`=` 赋值
`==` 不相等
`>` 大于
`<` 小于
`>=` 大于等于
`<=` 小于等于

逻辑运算符

```cpp

```

`&&` 与

```cpp
a && b  // a 和 b 都为真才为真
```

`||` 或

```cpp

```

5. 程序流程结构

顺序、选择、循环

选择结构

```cpp
if (条件1)
{
    // 条件为真时执行
}
else if (条件2)
{
    // 条件1不成立且条件2成立时执行
}
else
{
    // 上面条件都不成立时执行
}
```

三目运算符

格式：`表达式1 ? 表达式2 : 表达式3`

表达式1为真，结果为表达式2
表达式1为假，结果为表达式3

```cpp
int a = 10;
int b = 20;
int c = 0;

c = (a > b ? a : b);

cout << "b=" << b << endl;
cout << "c=" << c << endl;
```

`c = (a > b ? a : b)`，`c = 20`

switch语句

```cpp
switch (变量)
{
case A:
    // ...
    break;
case B:
    // ...
    break;
default:
    // 都不满足
    break;
}
```

`case` 后面只能是整型或字符型常量
每个 `case` 后一般写 `break`，否则会继续往下执行

6. 循环结构

while循环

```cpp
while (条件)
{
    // 条件为真时循环
}
```

特点：直到满足条件的最后才停止

do...while循环

```cpp
do
{
    // 循环体
} while (条件);
```

先做一次，再判断条件

for循环

```cpp
for (起始；条件；步长)
{
    // 循环体
}
```

```cpp
for (起始；条件；步长)
{
    for (起始；条件；步长)
    {
    }
}
```

内层循环执行完之后，才会执行外层循环的下一次

跳转语句

break

结束整个 `switch`
结束整个循环
嵌套循环中，结束最近的一层循环

continue

跳过本次循环剩下的语句，直接进行下一次循环

goto

```cpp
xxx
xxx
goto FLAG;
xxx
xxx
FLAG:
xxx
```

直接跳到标记处

值传递

形参的改变不影响实参

```cpp
void swap(int num1, int num2)
{
    int temp = num1;
    num1 = num2;
    num2 = temp;

    cout << num1 << "," << num2 << endl;
}

int main()
{
    int a = 10;
    int b = 20;

    swap(a, b);

    cout << a << "," << b << endl;
}
```

函数内输出 `20,10`
函数外还是 `10,20`

9. 指针

指针的定义

```cpp
int a = 10;
int* p;
p = &a;
```

`&` 取地址
`*` 解引用

```cpp
int* p = &a;
*p = 100;

cout << a << endl;
```

`p` 是地址，`*p` 是 p 指向地址中的值

指针所占空间：32位系统为 `4 bytes`，64位系统为 `8 bytes`

空指针和野指针

空指针：指向编号为 `0` 的空间

```cpp
int* p = NULL;
```

野指针：指向一个非法的内存空间

```cpp
int* p = (int*)0x1100;
```

空指针和野指针都不能访问

const修饰指针

```cpp
const int* p = &a;
```

指针指向的值不可以修改

```cpp
int* const p = &a;
```

指针指向不可以修改

```cpp
const int* const p = &a;
```

指针指向和值都不可以修改

指针和数组

```cpp
int* p = arr;
```

就是数组首地址

```cpp
p++;

for (int i = 0; i < 10; i++)
{
    cout << *p << endl;
}
```

指针与函数

地址传递：可以修改实参

```cpp
void swap02(int* p1, int* p2)
{
    int temp = *p1;
    *p1 = *p2;
    *p2 = temp;
}

int a = 10;
int b = 20;
swap02(&a, &b);

cout << "a = " << a << endl;
cout << "b = " << b << endl;
```

`a = 20`
`b = 10`

指针、数组、函数

```cpp
void bubblesort(int* arr, int len)
{
    for (int i = 0; i < 10; i++)
    {
        for (int j = 0; j < 10 - i - 1; j++)
        {
            if (arr[j] > arr[j + 1])
            {
                int a = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = a;
            }
        }
    }
}

void print(int* arr, int len)
{
    for (int i = 0; i < len; i++)
    {
        cout << arr[i] << endl;
    }
}

int main()
{
    int arr[10] = {4, 3, 6, 9, 1, 2, 10, 8, 7, 5};
    int* p = arr;
    int q = 10;

    bubblesort(p, q);
    print(p, q);
}
```

10. 结构体

结构体属于用户自定义的数据类型

```cpp
struct student
{
    int age;
    int score;
};
```

创建结构体变量

```cpp
struct student a;
a.age = 18;
a.score = 100;
```

```cpp
struct student b = {18, 100};
```

```cpp
struct student c;
c.age = 18;
c.score = 100;
```

结构体数组

```cpp
struct student a[2] =
{
    {18, 100},
    {19, 120}
};

a[0].age = 20;
a[0].score = 150;

for (int i = 0; i < 2; i++)
{
    cout << a[i].age << endl;
    cout << a[i].score << endl;
}
```

结构体指针

```cpp
struct student* p = &a;

cout << p->age << endl;
```

通过结构体指针访问成员

嵌套结构体

```cpp
struct student
{
    int age;
};

struct teacher
{
    struct student a;
};

struct teacher b;
b.a.age = 18;
```

结构体中 const

```cpp
void print(const student* a)
{
    cout << a->age << endl;
}

struct student a = {18, 100};
```

案例 1

```cpp
#include <iostream>
#include <string>
using namespace std;

struct hero
{
    string name;
    int age;
    string sex;
};

struct hero arr[5] =
{
    {"刘备", 23, "男"},
    {"关羽", 22, "男"},
    {"赵云", 21, "男"},
    {"张飞", 20, "男"},
    {"貂蝉", 19, "女"}
};

int len = 5;

void print(struct hero arr[], int len)
{
    for (int i = 0; i < len; i++)
    {
        cout << "姓名：" << arr[i].name << endl;
        cout << "年龄" << arr[i].age << endl;
        cout << "性别" << arr[i].sex << endl;
    }
}

void paixu(struct hero arr[], int len)
{
    for (int i = 0; i < 5; i++)
    {
        for (int j = 0; j < 5 - i - 1; j++)
        {
            if (arr[j].age < arr[j + 1].age)
            {
                struct hero a = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = a;
            }
        }
    }
}

int main()
{
    paixu(arr, len);
    print(arr, len);
    return 0;
}
```

案例 2

```cpp
#include <iostream>
#include <string>
#include <ctime>
#include <cstdlib>
using namespace std;

struct student
{
    string sname;
    int score;
};

struct teacher
{
    string tname;
    struct student sarry[5];
};

string nameseed = "ABCDE";

void fuzhi(struct teacher tarry[], int len)
{
    for (int i = 0; i < len; i++)
    {
        tarry[i].tname = "teacher_";
        tarry[i].tname += nameseed[i];

        for (int j = 0; j < 5; j++)
        {
            tarry[i].sarry[j].sname = "student_";
            tarry[i].sarry[j].sname += nameseed[j];
            int random = rand() % 61 + 40;
            tarry[i].sarry[j].score = random;
        }
    }
}

void print(struct teacher tarry[], int len)
{
    for (int i = 0; i < len; i++)
    {
        cout << tarry[i].tname << endl;
        for (int j = 0; j < 5; j++)
        {
            cout << tarry[i].sarry[j].sname << endl;
            cout << tarry[i].sarry[j].score << endl;
        }
    }
}

int main()
{
    srand((unsigned int)time(NULL));

    struct teacher tarry[3];
    int len = sizeof(tarry) / sizeof(tarry[0]);
    fuzhi(tarry, len);
    print(tarry, len);
    system("pause");
    return 0;
}
```

`rand() % 61` 取到 `0 ~ 60`
`rand() % 61 + 40` 取到 `40 ~ 100`
`srand((unsigned int)time(NULL))` 设置随机数种子
