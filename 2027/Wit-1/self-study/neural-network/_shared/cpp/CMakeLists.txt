cmake_minimum_required(VERSION 3.10)
project(neural_course LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio calib3d dnn)
get_filename_component(NN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(OLD "${NN_ROOT}/../opencv")
add_library(nn_detection "${OLD}/_shared/src/common.cpp" "${OLD}/_shared/src/detector.cpp")
target_include_directories(nn_detection PUBLIC ${OpenCV_INCLUDE_DIRS} "${OLD}/_shared/include"
  "${OLD}/02-hsv/src" "${OLD}/03-contours/src" "${OLD}/04-light-direction/src" "${OLD}/05-light-pairing/src")
target_link_libraries(nn_detection PUBLIC ${OpenCV_LIBS})
add_executable(nn_course "${NN_ROOT}/08-cpp-inference/src/main.cpp")
target_include_directories(nn_course PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(nn_course PRIVATE nn_detection)
target_compile_options(nn_course PRIVATE -Wall -Wextra)
