cmake_minimum_required(VERSION 3.10)
project(digit_classifier_demo)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS roscpp rospy sensor_msgs std_msgs geometry_msgs cv_bridge message_generation)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio calib3d dnn)
add_message_files(FILES DigitPrediction.msg)
generate_messages(DEPENDENCIES std_msgs geometry_msgs)
catkin_package(CATKIN_DEPENDS message_runtime roscpp sensor_msgs std_msgs geometry_msgs cv_bridge)
get_filename_component(PACKAGE_REAL "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
get_filename_component(NN_ROOT "${PACKAGE_REAL}/../.." ABSOLUTE)
set(OLD "${NN_ROOT}/../opencv")
add_executable(digit_node src/main.cpp "${OLD}/_shared/src/common.cpp" "${OLD}/_shared/src/detector.cpp")
target_include_directories(digit_node PRIVATE ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} "${NN_ROOT}/_shared/cpp"
  "${OLD}/_shared/include" "${OLD}/02-hsv/src" "${OLD}/03-contours/src" "${OLD}/04-light-direction/src" "${OLD}/05-light-pairing/src")
target_link_libraries(digit_node ${catkin_LIBRARIES} ${OpenCV_LIBS})
add_dependencies(digit_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
