cmake_minimum_required(VERSION 3.10)
project(opencv_offline_course LANGUAGES CXX)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio calib3d)
add_library(course_core src/common.cpp src/assets.cpp src/detector.cpp)
target_compile_features(course_core PUBLIC cxx_std_17)
target_compile_options(course_core PRIVATE -Wall -Wextra)
target_include_directories(course_core PUBLIC include ${OpenCV_INCLUDE_DIRS}
  ../02-hsv/src ../03-contours/src ../04-light-direction/src ../05-light-pairing/src)
target_link_libraries(course_core PUBLIC ${OpenCV_LIBS})
function(add_lesson target entry folder)
  add_executable(${target} src/entry.cpp ../${folder}/src/main.cpp ${ARGN})
  target_compile_definitions(${target} PRIVATE COURSE_ENTRY=${entry})
  target_compile_options(${target} PRIVATE -Wall -Wextra)
  target_link_libraries(${target} PRIVATE course_core)
endfunction()

target_link_libraries(course_assets PRIVATE course_core)
